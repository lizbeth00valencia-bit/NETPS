// Ejecutar esta función UNA SOLA VEZ para configurar los encabezados en Google Sheets function setupSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Respuestas de formulario 1") || ss.getSheets()[0]; var encabezados = [ "Marca Temporal", "Formato No.", "Fecha", "Hora", "Nombre Completo", "Cédula", "Duración Servicio", "Servicio Utilizado", "Escala Emoji", "Escala Numérica (NPS)", "Motivo de Calificación", "Firma Digital (URL)" ]; // Si la hoja está completamente vacía, agrega los encabezados con estilo corporativo if (sheet.getLastRow() === 0) { sheet.appendRow(encabezados); var rangoEncabezado = sheet.getRange(1, 1, 1, encabezados.length); rangoEncabezado .setFontFamily("Century Gothic") .setFontSize(10) .setFontWeight("bold") .setFontColor("#ffffff") .setBackground("#1e293b") .setHorizontalAlignment("center") .setVerticalAlignment("middle"); sheet.setRowHeight(1, 30); sheet.setFrozenRows(1); } } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Respuestas de formulario 1") || ss.getSheets()[0]; // Verificación e inserción automática de encabezados si no existen if (sheet.getLastRow() === 0) { setupSheet(); } var params = e.parameter; var urlFirma = "Sin firma"; // Calcular consecutivo automático basado en filas de datos (excluyendo encabezado) var totalFilas = sheet.getLastRow(); var numeroConsecutivo = totalFilas; // La Fila 1 es el encabezado, por ende la primera respuesta es 1 var codigoFormato = "FMT-" + ("00" + numeroConsecutivo).slice(-3); // Procesar firma digital en Drive if (params.firma && params.firma.indexOf("data:image") !== -1) { try { var folderName = "Firmas_Formularios"; var folders = DriveApp.getFoldersByName(folderName); var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); var base64Data = params.firma.split(",")[1]; var blob = Utilities.newBlob( Utilities.base64Decode(base64Data), "image/png", "Firma_" + codigoFormato + "_" + new Date().getTime() + ".png" ); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); urlFirma = file.getUrl(); } catch (err) { urlFirma = "Error al guardar imagen: " + err.toString(); } } var cedula = params.cedula || "Sin cédula"; // Guardar fila alineada con los encabezados sheet.appendRow([ new Date(), codigoFormato, params.fecha, params.hora, params.nombre, cedula, params.duracion, params.servicio, params.escalaEmoji, params.escalaNumero, params.comentarios, urlFirma ]); return ContentService.createTextOutput("Éxito").setMimeType(ContentService.MimeType.TEXT); }
